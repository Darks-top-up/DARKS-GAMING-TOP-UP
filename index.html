import os import psutil from kivy.app import App from kivy.uix.boxlayout import BoxLayout from kivy.uix.label import Label from kivy.uix.button import Button from kivy.uix.progressbar import ProgressBar from kivy.clock import Clock import kivy.utils # Import Android specific modules safely if running on Android if kivy.utils.platform == 'android': from jnius import autoclass PythonActivity = autoclass('org.kivy.android.PythonActivity') Intent = autoclass('android.content.Intent') PackageManager = autoclass('android.content.pm.PackageManager') class GameBoosterUI(BoxLayout): def __init__(self, **kwargs): super().__init__(**kwargs) self.orientation = 'vertical' self.padding = 20 self.spacing = 15 # Title self.title = Label( text="[b]Game Booster & Launcher[/b]", markup=True, font_size='20sp', size_hint_y=None, height=40 ) self.add_widget(self.title) # RAM Usage Display self.ram_label = Label(text="Checking RAM Usage...", font_size='16sp') self.add_widget(self.ram_label) self.ram_bar = ProgressBar(max=100) self.add_widget(self.ram_bar) # Action Button: Optimize Memory self.boost_btn = Button( text="Optimize Device Memory", background_color=(0, 0.9, 0.4, 1), size_hint_y=None, height=50 ) self.boost_btn.bind(on_press=self.run_optimization) self.add_widget(self.boost_btn) # Action Button: Launch PUBG Mobile self.launch_pubg_btn = Button( text="Launch PUBG Mobile", background_color=(0, 0.5, 1, 1), size_hint_y=None, height=50 ) self.launch_pubg_btn.bind(on_press=self.launch_pubg) self.add_widget(self.launch_pubg_btn) # Status Terminal self.status_label = Label( text="Status: Ready\n(Safe App Launcher & RAM Cleaner)", font_size='12sp', color=(0.7, 0.7, 0.7, 1) ) self.add_widget(self.status_label) # Auto-update stats every 2 seconds Clock.schedule_interval(self.update_ram_stats, 2.0) def update_ram_stats(self, dt=None): mem = psutil.virtual_memory() self.ram_bar.value = mem.percent self.ram_label.text = f"RAM Usage: {mem.percent}% ({round(mem.used/1024**3, 2)}GB / {round(mem.total/1024**3, 2)}GB)" def run_optimization(self, instance): self.status_label.text = "Optimizing: Clearing temporary cache..." try: import gc collected = gc.collect() self.update_ram_stats() self.status_label.text = f"Optimization Complete!\nFreed memory blocks: {collected}\nSystem ready." except Exception as e: self.status_label.text = f"Error during optimization: {str(e)}" def launch_pubg(self, instance): if kivy.utils.platform == 'android': try: activity = PythonActivity.mActivity pm = activity.getPackageManager() # Common package names for PUBG variants: # Global: com.tencent.ig # India (BGMI): com.pubg.imobile # KR/JP: com.pubg.krmobile packages_to_try = ["com.tencent.ig", "com.pubg.imobile", "com.pubg.krmobile"] launched = False for pkg in packages_to_try: intent = pm.getLaunchIntentForPackage(pkg) if intent is not None: activity.startActivity(intent) self.status_label.text = f"Successfully launched package: {pkg}" launched = True break if not launched: self.status_label.text = "Error: PUBG Mobile is not installed on this device." except Exception as e: self.status_label.text = f"Launch failed: {str(e)}" else: self.status_label.text = "Notice: App launching only works on Android devices, not PC." class GameBoosterApp(App): def build(self): return GameBoosterUI() if __name__ == "__main__": GameBoosterApp().run()
